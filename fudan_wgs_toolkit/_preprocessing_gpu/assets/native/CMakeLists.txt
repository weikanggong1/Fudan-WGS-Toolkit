cmake_minimum_required(VERSION 3.18)
project(head_native_pairhmm LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PHMM_ENABLE_CUDA "Build the real CUDA provider" ON)
add_library(pairhmm_oracle SHARED pairhmm_oracle.cpp)
target_include_directories(pairhmm_oracle PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
target_compile_options(pairhmm_oracle PRIVATE -Wall -Wextra -Werror -ffp-contract=off)
add_executable(pairhmm_selftest pairhmm_selftest.cpp)
target_compile_options(pairhmm_selftest PRIVATE -ffp-contract=off)
target_link_libraries(pairhmm_selftest PRIVATE pairhmm_oracle)
if(PHMM_ENABLE_CUDA)
    if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
        set(CMAKE_CUDA_ARCHITECTURES 80)
    endif()
    set(CMAKE_CUDA_RUNTIME_LIBRARY Shared)
    enable_language(CUDA)
    find_package(CUDAToolkit REQUIRED)
    add_library(pairhmm_cuda SHARED pairhmm_cuda.cu pairhmm_oracle.cpp)
    set_target_properties(pairhmm_cuda PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON)
    target_include_directories(pairhmm_cuda PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
    target_compile_options(pairhmm_cuda PRIVATE
        $<$<COMPILE_LANGUAGE:CUDA>:--fmad=false;--ftz=true;--prec-div=true;--prec-sqrt=true>
        $<$<COMPILE_LANGUAGE:CXX>:-ffp-contract=off;-Wall;-Wextra;-Werror>)
    target_link_libraries(pairhmm_cuda PRIVATE CUDA::cudart)
    target_compile_definitions(pairhmm_selftest PRIVATE PHMM_HAS_CUDA=1)
    target_link_libraries(pairhmm_selftest PRIVATE pairhmm_cuda)
    add_executable(pairhmm_public_fixture_test pairhmm_public_fixture_test.cpp)
    target_include_directories(pairhmm_public_fixture_test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR})
    target_link_libraries(pairhmm_public_fixture_test PRIVATE pairhmm_cuda)
endif()
